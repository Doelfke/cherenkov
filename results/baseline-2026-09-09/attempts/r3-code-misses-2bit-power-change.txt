class ListNode:
    """Definition for singly-linked list node."""
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next


def reverse_linked_list(head: ListNode) -> ListNode:
    """
    Reverses a singly linked list and returns the new head.
    
    Args:
        head: The head node of the linked list.
        
    Returns:
        The new head node of the reversed linked list.
    """
    prev = None
    current = head
    
    while current:
        next_node = current.next  # Store the next node
        current.next = prev       # Reverse the link
        prev = current            # Move prev forward
        current = next_node       # Move current forward
    
    return prev  # prev is the new head


def create_linked_list(values: list) -> ListNode:
    """Helper function to create a linked list from a list of values."""
    if not values:
        return None
    head = ListNode(values[0])
    current = head
    for val in values[1:]:
        current.next = ListNode(val)
        current = current.next
    return head


def linked_list_to_list(head: ListNode) -> list:
    """Helper function to convert a linked list to a Python list."""
    result = []
    current = head
    while current:
        result.append(current.val)
        current = current.next
    return result


if __name__ == "__main__":
    # Test case 1: Normal list
    head = create_linked_list([1, 2, 3, 4, 5])
    reversed_head = reverse_linked_list(head)
    print(linked_list_to_list(reversed_head))  # Expected: [5, 4, 3, 2, 1]
    
    # Test case 2: Empty list
    head = create_linked_list([])
    reversed_head = reverse_linked_list(head)
    print(linked_list_to_list(reversed_head))  # Expected: []
    
    # Test case 3: Single element
    head = create_linked_list([1])
    reversed_head = reverse_linked_list(head)
    print(linked_list_to_list(reversed_head))  # Expected: [1]
    
    # Test case 4: Two elements
    head = create_linked_list([1, 2])
    reversed_head = reverse_linked_list(head)
    print(linked_list_to_list(reversed_head))  # Expected: [2, 1]
